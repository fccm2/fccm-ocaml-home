open Magick
let () =
  let color = (65_000, 1_000, 0, 65_535) in
  let img = Magick.new_image 210 120 (color) in
  Magick.image_display img;
;;

And here is how to draw your first primitive:

open Magick
let () =
  let color = (65_000, 1_000, 0, 65_535) in
  let img = Magick.new_image 210 120 (color) in
  let prim = Magick.Prim.draw_rectangle (10, 10) (100, 60) in
  let color = Magick.Color.map8 (0, 0, 0, 255) in
  Magick.fill_primitive img ~prim:prim ~fill:color ();
  Magick.image_display img;
;;
